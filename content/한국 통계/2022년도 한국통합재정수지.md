## 2022회계연도 정부운영표

<div id="datatable-root"></div>

<script type="module">
document.addEventListener('DOMContentLoaded', async function() {
  const csvPath = '/static/statement_of_operations2022.csv';
  const container = document.getElementById('datatable-root');
  
  // CSV 파싱
  function parseCSV(text) {
    const lines = text.trim().split('\n');
    const result = [];
    for (let line of lines) {
      const row = [];
      let current = '';
      let inQuotes = false;
      for (let i = 0; i < line.length; i++) {
        const char = line[i];
        if (char === '"') {
          inQuotes = !inQuotes;
        } else if (char === ',' && !inQuotes) {
          row.push(current.trim());
          current = '';
        } else {
          current += char;
        }
      }
      row.push(current.trim());
      result.push(row);
    }
    return result;
  }
  
  // 숫자 감지 및 포맷팅
  function formatNumber(value) {
    const cleaned = value.replace(/[,"]/g, '');
    if (!isNaN(cleaned) && cleaned !== '') {
      return parseFloat(cleaned).toLocaleString('ko-KR');
    }
    return value;
  }
  
  // 계층 레벨 감지
  function detectLevel(text) {
    const match = text.match(/^(\s+)/);
    return match ? Math.floor(match[1].length / 2) : 0;
  }
  
  try {
    const response = await fetch(csvPath);
    const csvText = await response.text();
    const rows = parseCSV(csvText);
    
    // 테이블 생성
    let html = '<div style="overflow-x:auto;border:1px solid #ddd;border-radius:8px;margin:2rem 0;"><table style="width:100%;border-collapse:collapse;font-size:0.9rem;"><thead style="background:#f8f9fa;position:sticky;top:0;"><tr>';
    
    // 헤더
    rows[0].forEach(header => {
      html += `<th style="padding:12px;text-align:center;border-bottom:2px solid #ddd;">${header}</th>`;
    });
    html += '</tr></thead><tbody>';
    
    // 데이터
    for (let i = 1; i < rows.length; i++) {
      const level = detectLevel(rows[i][0]);
      const bgColor = level === 0 ? 'rgba(59,130,246,0.05)' : level === 1 ? 'rgba(59,130,246,0.02)' : '';
      html += `<tr style="background:${bgColor};border-bottom:1px solid #eee;">`;
      
      rows[i].forEach((cell, j) => {
        const align = j === 0 ? 'left' : 'right';
        const padding = j === 0 ? `${level * 1.5 + 1}rem` : '0.75rem';
        const value = j === 0 ? cell.trim() : formatNumber(cell);
        html += `<td style="padding:10px;padding-left:${padding};text-align:${align};">${value}</td>`;
      });
      
      html += '</tr>';
    }
    
    html += '</tbody></table></div>';
    container.innerHTML = html;
    
  } catch (error) {
    container.innerHTML = `<div style="padding:2rem;text-align:center;color:#666;">⚠️ 데이터를 불러올 수 없습니다: ${error.message}</div>`;
  }
});
</script>
